function encodeBase64(){
const input=document.getElementById('enc-input').value;
if(!input){showToast('أدخل نصاً أولاً','error');return}
try{const encoded=btoa(unescape(encodeURIComponent(input)));document.getElementById('enc-result').textContent=encoded;document.getElementById('enc-output').style.display='block';showToast('تم الترميز بنجاح','success')}catch(e){showToast('خطأ في الترميز: '+e.message,'error')}
}
function decodeBase64(){
const input=document.getElementById('enc-input').value;
if(!input){showToast('أدخل نصاً أولاً','error');return}
try{const decoded=decodeURIComponent(escape(atob(input)));document.getElementById('enc-result').textContent=decoded;document.getElementById('enc-output').style.display='block';showToast('تم فك الترميز بنجاح','success')}catch(e){showToast('خطأ في فك الترميز: '+e.message,'error')}
}

async function generateHashes(){
const input=document.getElementById('hash-input').value;
if(!input){document.getElementById('hash-md5').textContent='---';document.getElementById('hash-sha1').textContent='---';document.getElementById('hash-sha256').textContent='---';document.getElementById('hash-sha512').textContent='---';return}
const encoder=new TextEncoder();
const data=encoder.encode(input);
document.getElementById('hash-md5').textContent=await simpleHash(data,'MD5');
document.getElementById('hash-sha1').textContent=await simpleHash(data,'SHA-1');
document.getElementById('hash-sha256').textContent=await simpleHash(data,'SHA-256');
document.getElementById('hash-sha512').textContent=await simpleHash(data,'SHA-512');
}

async function simpleHash(data,algo){
try{const hashBuffer=await crypto.subtle.digest(algo,data);const hashArray=Array.from(new Uint8Array(hashBuffer));return hashArray.map(b=>b.toString(16).padStart(2,'0')).join('')}catch(e){return 'Error'}
}

function downloadFile(filename,content){
const blob=new Blob([content],{type:'text/plain'});
const url=URL.createObjectURL(blob);
const a=document.createElement('a');
a.href=url;
a.download=filename;
document.body.appendChild(a);
a.click();
document.body.removeChild(a);
URL.revokeObjectURL(url);
showToast('تم تحميل '+filename,'success');
}

function navigateCommands(e){
const items=document.querySelectorAll('.command-item');
if(e.key==='ArrowDown'){e.preventDefault();selectedCmdIndex=(selectedCmdIndex+1)%items.length;updateSelection(items)}
else if(e.key==='ArrowUp'){e.preventDefault();selectedCmdIndex=(selectedCmdIndex-1+items.length)%items.length;updateSelection(items)}
else if(e.key==='Enter'){e.preventDefault();if(items[selectedCmdIndex])items[selectedCmdIndex].click()}
else if(e.key==='Escape'){document.getElementById('commandPalette').classList.remove('active')}
}

function updateSelection(items){
items.forEach((item,i)=>item.classList.toggle('selected',i===selectedCmdIndex));
if(items[selectedCmdIndex])items[selectedCmdIndex].scrollIntoView({block:'nearest'});
}

function executeCommand(target,type){
document.getElementById('commandPalette').classList.remove('active');
if(type==='action'&&target==='toggleTheme'){toggleTheme()}
else{showPage(target)}
}
